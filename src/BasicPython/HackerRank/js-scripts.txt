var languages = $('input[id^=enable-checkbox-]:checked').map(function () { return $(this).attr('id').substr('enable-checkbox-'.length)}).toArray();
var langIdx = 0;
function setTimeLimit(challengeId, timeLimit) {
    setTimeout(function () {
        new HR.Administration_ChallengeLanguageModel({
            "time_limit": timeLimit,
            "challenge_id": challengeId,
            "language_key": languages[langIdx]
        }).save();
        langIdx++;
        if (langIdx < languages.length) {
            setTimeLimit(challengeId, timeLimit);
        }
    }, 1000);    
}
langIdx = 0;
